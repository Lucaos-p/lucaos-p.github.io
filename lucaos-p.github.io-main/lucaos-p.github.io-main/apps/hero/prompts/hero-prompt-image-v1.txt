        // Build Palette UI
        function buildPaletteGrid() {
            let html = `<div class="palette-row">
                <span class="palette-row-label font-bold">Shade</span>
                <div class="palette-swatches">`;
            shadeKeys.forEach(shade => {
                html += `<span class="size-7 text-center text-xs font-bold text-slate-500 flex items-center justify-content">${shade}</span>`;
            });
            html += `</div></div>`;

            colorFamilies.forEach(family => {
                html += `<div class="palette-row">
                    <span class="palette-row-label">${family.name}</span>
                    <div class="palette-swatches">`;
                
                if (family.name === "black" || family.name === "white") {
                    const colorName = family.name;
                    const hex = family.shades.DEFAULT;
                    html += `<button type="button" class="swatch" style="background-color: ${hex};" data-color-name="${colorName}" aria-label="${colorName}" title="${colorName}"></button>`;
                } else {
                    shadeKeys.forEach(shade => {
                        const colorName = `${family.name}-${shade}`;
                        const hex = family.shades[shade];
                        html += `<button type="button" class="swatch" style="background-color: ${hex};" data-color-name="${colorName}" aria-label="${colorName}" title="${colorName}"></button>`;
                    });
                }
                html += `</div></div>`;
            });

            paletteGrid.innerHTML = html;

            paletteGrid.querySelectorAll(".swatch").forEach(btn => {
                btn.addEventListener("click", () => {
                    const colorName = btn.getAttribute("data-color-name");
                    applyColorToActiveTarget(colorName);
                });
            });
        }


        // 3. Read settings values

        function readSettings() {
            settings.darkAppearance = darkAppearance.checked;
            settings.centreText = centreText.checked;
            settings.minimumHeight = minimumHeight.checked;
            settings.showSubheading = showSubheading.checked;
            settings.gradientBackground = gradientBackground.checked;
            settings.gradientDirection = gradientDirection.value;

            settings.h1Serif = h1Serif.checked;
            settings.h1Spacing = h1Spacing.value + "em";
            h1SpacingVal.textContent = settings.h1Spacing;

            settings.h2Serif = h2Serif.checked;
            settings.h2Spacing = h2Spacing.value + "em";
            h2SpacingVal.textContent = settings.h2Spacing;
        }


        // Apply color selection helper
        function applyColorToActiveTarget(colorName) {
            if (settings.activeColorTarget === "background") {
                settings.bgColor = colorName;
            } else if (settings.activeColorTarget === "startColor") {
                settings.startColor = colorName;
            } else if (settings.activeColorTarget === "endColor") {
                settings.endColor = colorName;
            } else if (settings.activeColorTarget === "h1") {
                settings.h1Color = colorName;
            } else if (settings.activeColorTarget === "h2") {
                settings.h2Color = colorName;
                settings.savedH2Color = colorName;
            }
            updateHero();
        }


        // Map Tailwind color name to CSS value or variable
        function getColorValue(colorName) {
            if (colorName === "black") return "#000000";
            if (colorName === "white") return "#ffffff";
            const parts = colorName.split("-");
            if (parts.length === 2) {
                const family = colorFamilies.find(f => f.name === parts[0]);
                if (family && family.shades[parts[1]]) {
                    return family.shades[parts[1]];
                }
            }
            return colorName;
        }


        // 4. Update hero preview

        function updateHero() {
            readSettings();

            hero.classList.remove("hero-light", "hero-dark");
            hero.classList.add(
                settings.darkAppearance ? "hero-dark" : "hero-light"
            );

            hero.classList.toggle(
                "hero-centre",
                settings.centreText
            );

            hero.classList.toggle(
                "hero-min-height",
                settings.minimumHeight
            );

            // Toggle gradient/solid UI controls
            if (settings.gradientBackground) {
                targetBtnBg.classList.add("hidden");
                targetBtnStart.classList.remove("hidden");
                targetBtnEnd.classList.remove("hidden");
                directionGroup.classList.remove("hidden");
                directionGroup.classList.add("inline-flex");

                // If active target was background, switch to startColor
                if (settings.activeColorTarget === "background") {
                    settings.activeColorTarget = "startColor";
                    updateTargetButtonsUI();
                }
            } else {
                targetBtnBg.classList.remove("hidden");
                targetBtnStart.classList.add("hidden");
                targetBtnEnd.classList.add("hidden");
                directionGroup.classList.remove("inline-flex");
                directionGroup.classList.add("hidden");

                // If active target was startColor or endColor, switch to background
                if (settings.activeColorTarget === "startColor" || settings.activeColorTarget === "endColor") {
                    settings.activeColorTarget = "background";
                    updateTargetButtonsUI();
                }
            }

            // H1 & H2 Serif classes
            heroHeading.classList.toggle("hero-h1-serif", settings.h1Serif);
            heroSubheading.classList.toggle("hero-h2-serif", settings.h2Serif);

            // Subheading visibility & control states
            heroSubheading.hidden = !settings.showSubheading;
            
            const h2Inputs = h2ControlsGroup.querySelectorAll("input");
            h2Inputs.forEach(input => {
                input.disabled = !settings.showSubheading;
            });
            h2ControlsGroup.style.opacity = settings.showSubheading ? "1" : "0.5";
            h2TargetBtn.disabled = !settings.showSubheading;
            h2TargetBtn.style.opacity = settings.showSubheading ? "1" : "0.5";

            if (!settings.showSubheading && settings.activeColorTarget === "h2") {
                settings.activeColorTarget = settings.gradientBackground ? "startColor" : "background";
                updateTargetButtonsUI();
            }

            // Apply interactive letter spacings live
            heroHeading.style.letterSpacing = settings.h1Spacing;
            heroSubheading.style.letterSpacing = settings.h2Spacing;

            // Apply custom custom colors & gradient via CSS variables
            if (settings.gradientBackground) {
                hero.style.setProperty("--hero-bg", "transparent");
                hero.style.setProperty("--hero-bg-image", `linear-gradient(${settings.gradientDirection}, ${getColorValue(settings.startColor)}, ${getColorValue(settings.endColor)})`);
            } else {
                hero.style.setProperty("--hero-bg", getColorValue(settings.bgColor));
                hero.style.setProperty("--hero-bg-image", "none");
            }

            heroHeading.style.color = getColorValue(settings.h1Color);
            heroSubheading.style.color = getColorValue(settings.h2Color);

            // Update UI indicator text for targets
            bgCurrentColorName.textContent = settings.bgColor;
            startCurrentColorName.textContent = settings.startColor;
            endCurrentColorName.textContent = settings.endColor;
            h1CurrentColorName.textContent = settings.h1Color;
            h2CurrentColorName.textContent = settings.h2Color;

            // Update active swatch highlights
            updateSwatchHighlights();

            // Update dynamic style element in head for export/preview consistency
            updateDynamicStyles();

            updateGeneratedHTML();
        }


        function updateTargetButtonsUI() {
            targetButtons.forEach(btn => {
                const target = btn.getAttribute("data-target");
                if (target === settings.activeColorTarget) {
                    btn.classList.add("active");
                } else {
                    btn.classList.remove("active");
                }
            });
        }


        function updateSwatchHighlights() {
            let activeColor = settings.bgColor;
            if (settings.activeColorTarget === "startColor") activeColor = settings.startColor;
            if (settings.activeColorTarget === "endColor") activeColor = settings.endColor;
            if (settings.activeColorTarget === "h1") activeColor = settings.h1Color;
            if (settings.activeColorTarget === "h2") activeColor = settings.h2Color;

            paletteGrid.querySelectorAll(".swatch").forEach(swatch => {
                if (swatch.getAttribute("data-color-name") === activeColor) {
                    swatch.classList.add("selected");
                } else {
                    swatch.classList.remove("selected");
                }
            });
        }


        function updateDynamicStyles() {
            const styleElem = document.getElementById("dynamicHeroStyles");
            let bgRule = `background-color: ${getColorValue(settings.bgColor)} !important;`;
            let bgImgRule = `background-image: none !important;`;
            if (settings.gradientBackground) {
                bgRule = `background-color: transparent !important;`;
                bgImgRule = `background-image: linear-gradient(${settings.gradientDirection}, ${getColorValue(settings.startColor)}, ${getColorValue(settings.endColor)}) !important;`;
            }

            styleElem.textContent = `
        #heroHeading {
            letter-spacing: ${settings.h1Spacing};
        }
        #heroSubheading {
            letter-spacing: ${settings.h2Spacing};
        }
        #hero {
            ${bgRule}
            ${bgImgRule}
        }
        #heroHeading {
            color: ${getColorValue(settings.h1Color)} !important;
        }
        #heroSubheading {
            color: ${getColorValue(settings.h2Color)} !important;
        }`;
        }


        // 5. Escape special HTML characters

        function escapeHTML(text) {
            return text
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;");
        }

        // 7. Event listeners for inputs and controls

        [
            darkAppearance,
            centreText,
            minimumHeight,
            showSubheading,
            gradientBackground
        ].forEach(checkbox => {
            checkbox.addEventListener("change", () => {
                if (checkbox === darkAppearance) {
                    if (darkAppearance.checked) {
                        settings.bgColor = "slate-900";
                        settings.h1Color = "slate-50";
                        settings.h2Color = "slate-300";
                        settings.savedH2Color = "slate-300";
                    } else {
                        settings.bgColor = "slate-50";
                        settings.h1Color = "slate-900";
                        settings.h2Color = "slate-700";
                        settings.savedH2Color = "slate-700";
                    }
                }
                if (checkbox === gradientBackground) {
                    if (gradientBackground.checked) {
                        settings.startColor = settings.bgColor;
                        settings.endColor = "blue-500";
                        settings.activeColorTarget = "startColor";
                    } else {
                        settings.bgColor = settings.startColor;
                        settings.activeColorTarget = "background";
                    }
                    updateTargetButtonsUI();
                }
                updateHero();
            });
        });

        gradientDirection.addEventListener("change", updateHero);

        [h1Spacing, h2Spacing].forEach(slider => {
            slider.addEventListener("input", updateHero);
        });

        [h1Serif, h2Serif].forEach(checkbox => {
            checkbox.addEventListener("change", updateHero);
        });

        [heroHeading, heroSubheading].forEach(elem => {
            elem.addEventListener("input", updateGeneratedHTML);
        });

        targetButtons.forEach(btn => {
            btn.addEventListener("click", () => {
                const target = btn.getAttribute("data-target");
                if (target === "h2" && !settings.showSubheading) return;
                settings.activeColorTarget = target;
                updateTargetButtonsUI();
                updateSwatchHighlights();
            });
        });


        // 8. Clipboard Copy functionality

        async function handleCopy() {
            const htmlString = createExportHTML();
            try {
                if (navigator.clipboard && navigator.clipboard.writeText) {
                    await navigator.clipboard.writeText(htmlString);
                    fallbackPanel.hidden = true;
                    copyStatus.textContent = "HTML copied—paste it into VS Code.";
                } else {
                    throw new Error("Clipboard API unavailable");
                }
            } catch (err) {
                generatedHTML.value = htmlString;
                fallbackPanel.hidden = false;
                generatedHTML.select();
                copyStatus.textContent = "Copying failed. Please copy the HTML manually from the box below.";
            }
        }

        copyButton.addEventListener("click", handleCopy);

        updateTargetButtonsUI();
        updateHero();
